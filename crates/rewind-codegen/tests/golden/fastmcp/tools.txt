== notes.search_notes
notes.search_notes({ query: string, limit?: number, sort?: "relevance" | "newest" | "oldest", tags?: string[] | null }) → { result: { body: string | null, created: string, id: string, tags: string[], title: string }[] } · Search notes by text and tags.

declare module "rewind:tools" {
  /** Tools of the MCP server `notes`. */
  export const notes: {
    /** Search notes by text and tags. */
    search_notes(args: {
      query: string;
      /**
       * @maximum 100
       * @default 20
       */
      limit?: number;
      /** @default "relevance" */
      sort?: notes.Sort;
      /** @default null */
      tags?: string[] | null;
    }): Promise<{ result: notes.Note[] }>;
  };
  export namespace notes {
    export type Sort = "relevance" | "newest" | "oldest";

    export interface Note {
      body: string | null;
      /** @format date-time */
      created: string;
      id: string;
      tags: string[];
      title: string;
    }
  }
}
